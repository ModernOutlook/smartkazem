// ===== ابزارهای اعتبارسنجی =====
function countWords(text) {
  return text.trim().split(/\s+/).filter(Boolean).length;
}

function countChars(text) {
  return text.replace(/\s+/g, " ").trim().length;
}

function wordLength(word) {
  const cleaned = word.replace(/^[.,;:!?«»"'()\[\]{}…\-—–]+|[.,;:!?«»"'()\[\]{}…\-—–]+$/g, "");
  return cleaned.length;
}

function findLongWords(text) {
  const words = text.trim().split(/\s+/);
  return words.filter(w => wordLength(w) > MAX_WORD_LENGTH);
}

function breakLongWords(text) {
  return text.split(/\s+/).map(w => {
    if (wordLength(w) <= MAX_WORD_LENGTH) return w;
    let pieces = [w];
    let guard = 0;
    while (pieces.some(p => p.length > MAX_WORD_LENGTH) && guard < 20) {
      pieces = pieces.flatMap(p => {
        if (p.length <= MAX_WORD_LENGTH) return [p];
        const half = Math.ceil(p.length / 2);
        return [p.slice(0, half), p.slice(half)];
      });
      guard++;
    }
    return pieces.join("‌");
  }).join(" ");
}

function enforceWordLimit(text, maxWords) {
  const words = text.trim().split(/\s+/);
  if (words.length <= maxWords) return { text, truncated: false };
  return { text: words.slice(0, maxWords).join(" "), truncated: true };
}

function enforceCharLimit(text, maxChars) {
  if (text.length <= maxChars) return { text, truncated: false };
  let cut = text.slice(0, maxChars);
  const lastSpace = cut.lastIndexOf(" ");
  if (lastSpace > maxChars * 0.7) cut = cut.slice(0, lastSpace);
  return { text: cut.trim(), truncated: true };
}

function processText(text) {
  const originalWords = countWords(text);
  const originalChars = countChars(text);
  const longWords = findLongWords(text);

  let processed = text;
  let truncatedByWord = false;
  let truncatedByChar = false;

  if (longWords.length > 0) processed = breakLongWords(processed);

  const wordLock = enforceWordLimit(processed, MAX_WORDS);
  processed = wordLock.text;
  truncatedByWord = wordLock.truncated;

  const charLock = enforceCharLimit(processed, MAX_CHARS);
  processed = charLock.text;
  truncatedByChar = charLock.truncated;

  const finalWords = countWords(processed);
  const finalChars = countChars(processed);
  const finalLongWords = findLongWords(processed);

  return {
    text: processed,
    originalWordCount: originalWords,
    originalCharCount: originalChars,
    finalWordCount: finalWords,
    finalCharCount: finalChars,
    longWordsFound: longWords,
    remainingLongWords: finalLongWords,
    truncated: truncatedByWord || truncatedByChar,
    truncatedByWord,
    truncatedByChar,
    violations: {
      wordLimit: finalWords > MAX_WORDS,
      charLimit: finalChars > MAX_CHARS,
      wordLength: finalLongWords.length > 0
    }
  };
}

// ===== شمارنده‌ی زنده =====
function renderCounter(proc) {
  const badges = [];

  const wordCls = proc.violations.wordLimit ? "violation" : "okflag";
  badges.push(`<span class="${wordCls}">کلمات: ${proc.finalWordCount} / ${MAX_WORDS}</span>`);

  const charCls = proc.violations.charLimit ? "violation" : "okflag";
  badges.push(`<span class="${charCls}">حروف: ${proc.finalCharCount} / ${MAX_CHARS}</span>`);

  const longCls = proc.violations.wordLength ? "violation" : "okflag";
  badges.push(`<span class="${longCls}">کلمات بلند: ${proc.remainingLongWords.length}</span>`);

  if (proc.longWordsFound.length > 0) {
    badges.push(`<span class="violation">اصلاح‌شده: ${proc.longWordsFound.length} کلمه‌ی بلند</span>`);
  }
  if (proc.truncated) {
    const how = [];
    if (proc.truncatedByWord) how.push("کلمه");
    if (proc.truncatedByChar) how.push("حرف");
    badges.push(`<span class="violation">کوتاه‌سازی: ${how.join(" + ")}</span>`);
  }

  return `<div class="meta">${badges.join("")}</div>`;
}

// ===== رندر پاراگراف =====
function renderParagraph(p, index) {
  const j = p.judgment;
  const proc = processText(p.text);

  const rows = ["S", "T", "E", "R", "C"].map(k => {
    const s = statusMap[j[k].status];
    return `<tr>
      <td>${domainNames[k]}</td>
      <td class="status ${s.cls}">${s.label}</td>
      <td>${j[k].reason || ""}</td>
    </tr>`;
  }).join("");

  const counts = { ok: 0, warn: 0, bad: 0 };
  ["S","T","E","R","C"].forEach(k => counts[j[k].status]++);

  return `<div class="paragraph">
    <div class="paragraph-text">${proc.text}</div>
    ${renderCounter(proc)}
    <table>
      <thead><tr><th>قلمرو</th><th>وضعیت</th><th>دلیل</th></tr></thead>
      <tbody>${rows}</tbody>
    </table>
    <div class="summary">
      <span>✅ ${counts.ok}</span>
      <span>⚠️ ${counts.warn}</span>
      <span>❌ ${counts.bad}</span>
    </div>
  </div>`;
}

  btn.innerHTML = 'در حال تولید...<span class="spinner"></span>';
  $("output").innerHTML = "";

  const userPrompt = `تعداد پاراگراف: ${count}
${MODE_INSTRUCTIONS[mode]}

یادآوری محدودیت‌ها:
- هر پاراگراف حداکثر ${MAX_WORDS} کلمه.
- هر کلمه حداکثر ${MAX_WORD_LENGTH} حرف.
- مجموع حروف هر پاراگراف حداکثر ${MAX_CHARS}.

پاراگراف‌ها را تولید کن و ماتریس داوری هر کدام را طبق فرمت JSON بده.`;

  try {
    const res = await fetch(baseUrl + "/chat/completions", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Authorization": "Bearer " + apiKey
      },
      body: JSON.stringify({
        model: model,
        messages: [
          { role: "system", content: SYSTEM_PROMPT },
          { role: "user", content: userPrompt }
        ],
        temperature: 0.9,
        response_format: { type: "json_object" }
      })
    });

    if (!res.ok) {
      const t = await res.text();
      throw new Error(`خطای API (${res.status}): ${t.slice(0, 200)}`);
    }

    const data = await res.json();
    let content = data.choices[0].message.content.trim();
    content = content.replace(/^```json\s*/, "").replace(/```$/, "").trim();
    const parsed = JSON.parse(content);

    if (!parsed.paragraphs || !Array.isArray(parsed.paragraphs)) {
      throw new Error("ساختار پاسخ نامعتبر است.");
    }

    $("output").innerHTML = parsed.paragraphs.map(renderParagraph).join("");
  } catch (e) {
    $("error").textContent = "خطا: " + e.message;
  } finally {
    btn.disabled = false;
    btn.textContent = "تولید پاراگراف";
  }
}

$("generateBtn").addEventListener("click", generate);
